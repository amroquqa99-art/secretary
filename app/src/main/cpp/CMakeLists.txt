cmake_minimum_required(VERSION 3.22.1)
project(secretary_llama LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_RPC OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
if(ANDROID AND ANDROID_ABI STREQUAL "x86_64")
    set(GGML_AVX OFF CACHE BOOL "" FORCE)
    set(GGML_AVX2 OFF CACHE BOOL "" FORCE)
    set(GGML_FMA OFF CACHE BOOL "" FORCE)
    set(GGML_F16C OFF CACHE BOOL "" FORCE)
endif()
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
if(LLAMA_CPP_DIR)
    add_subdirectory("${LLAMA_CPP_DIR}" llama)
else()
    include(FetchContent)
    FetchContent_Declare(llama_cpp
        URL "https://codeload.github.com/ggml-org/llama.cpp/tar.gz/50569eb87df530daff11afda229ceb9ab8e6cae8?secretary=50569eb87df530daff11afda229ceb9ab8e6cae8"
        URL_HASH SHA256=10af4d51ee5c29566e1f2899752d12ea41004554ceff7a91ca931b0ace48bc93)
    FetchContent_MakeAvailable(llama_cpp)
endif()
add_library(secretary_llama SHARED core.cpp native_bridge.cpp)
target_link_libraries(secretary_llama PRIVATE llama)
if(NOT ANDROID)
    target_include_directories(secretary_llama PRIVATE "$ENV{JAVA_HOME}/include" "$ENV{JAVA_HOME}/include/linux")
endif()
